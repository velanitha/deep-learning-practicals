011. Dense neural network for MNIST; compare activation functions
Copy this code into a Google Colab cell.
import tensorflow as tf
from tensorflow.keras import layers, models
import matplotlib.pyplot as plt

# Load handwritten digit images (28x28 pixels)
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
# Scale pixel values from 0-255 to 0-1
x_train, x_test = x_train / 255.0, x_test / 255.0

# Train a small model for each activation function
for activation in ["relu", "tanh", "sigmoid"]:
    model = models.Sequential([
        layers.Flatten(input_shape=(28, 28)),
        layers.Dense(64, activation=activation),
        layers.Dense(10, activation="softmax")
    ])
    model.compile(optimizer="adam",
                  loss="sparse_categorical_crossentropy",
                  metrics=["accuracy"])
    model.fit(x_train[:10000], y_train[:10000], epochs=2, verbose=0)
    loss, accuracy = model.evaluate(x_test[:2000], y_test[:2000], verbose=0)
    print(activation, "test accuracy:", round(accuracy, 4))

from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from tensorflow.keras import Sequential, layers

# Load the Iris flower dataset
X, y = load_iris(return_X_y=True)

# Compare one hidden layer with two hidden layers
for hidden_layers in [1, 2, 3]:
    model = Sequential([layers.Input(shape=(4,))])
    for _ in range(hidden_layers):
        model.add(layers.Dense(16, activation="relu"))
    model.add(layers.Dense(3, activation="softmax"))
    model.compile(optimizer="adam", loss="sparse_categorical_crossentropy",
                  metrics=["accuracy"])
    model.fit(X_train, y_train, epochs=40, verbose=0)
    print(hidden_layers, "hidden layers - test accuracy:",
          round(model.evaluate(X_test, y_test, verbose=0)[1], 3))
033. Compare a basic CNN and a deeper CNN

import tensorflow as tf
from tensorflow.keras import layers, models

# MNIST images are small and quick to train
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
x_train = x_train[:10000, ..., None] / 255.0
x_test = x_test[:2000, ..., None] / 255.0
y_train, y_test = y_train[:10000], y_test[:2000]

# Build and test two CNN designs
for deep in [False, True]:
    model = models.Sequential([layers.Input((28, 28, 1)),
                               layers.Conv2D(16, 3, activation="relu"),
                               layers.MaxPooling2D()])
    if deep:
        model.add(layers.Conv2D(32, 3, activation="relu"))
        model.add(layers.MaxPooling2D())
    model.add(layers.Flatten())
    model.add(layers.Dense(32, activation="relu"))
    model.add(layers.Dense(10, activation="softmax"))
    model.compile(optimizer="adam", loss="sparse_categorical_crossentropy",
                  metrics=["accuracy"])
    model.fit(x_train, y_train, epochs=2, verbose=0)
    print("Deeper CNN" if deep else "Basic CNN",
          "accuracy:", round(model.evaluate(x_test, y_test, verbose=0)[1], 3))

import tensorflow as tf
from tensorflow.keras import layers, models

# Use a small built-in movie-review dataset
vocab_size, max_len = 10000, 100
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.imdb.load_data(num_words=vocab_size)

088. Autoencoder for MNIST compression and reconstruction
Copy this code into a Google Colab cell.
import tensorflow as tf
from tensorflow.keras import layers, models
import matplotlib.pyplot as plt

(x_train, _), (x_test, _) = tf.keras.datasets.mnist.load_data()
x_train = x_train[:5000].astype("float32") / 255.0
x_test = x_test[:5].astype("float32") / 255.0

encoder_input = layers.Input((28, 28))
x = layers.Flatten()(encoder_input)
encoded = layers.Dense(32, activation="relu")(x)  # compressed representation
x = layers.Dense(784, activation="sigmoid")(encoded)
decoded = layers.Reshape((28, 28))(x)
autoencoder = models.Model(encoder_input, decoded)
autoencoder.compile(optimizer="adam", loss="mse")
autoencoder.fit(x_train, x_train, epochs=3, batch_size=64, verbose=1)

reconstructed = autoencoder.predict(x_test, verbose=0)
for i in range(5):
    plt.subplot(2, 5, i+1); plt.imshow(x_test[i], cmap="gray"); plt.axis("off")
    plt.subplot(2, 5, i+6); plt.imshow(reconstructed[i], cmap="gray"); plt.axis("off")
plt.show()

1111. Siamese network: decide if two MNIST digits are the same
Copy this code into a Google Colab cell.
import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, models

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
x_train = x_train[:10000].astype("float32") / 255.0
y_train = y_train[:10000]
x_test = x_test[:2000].astype("float32") / 255.0
y_test = y_test[:2000]

# Create pairs: half same class, half different class
def make_pairs(images, labels, n=2000):
    a, b, targets = [], [], []
    for i in range(n):
        idx1 = np.random.randint(len(images))
        same = i % 2 == 0
        candidates = np.where(labels == labels[idx1])[0] if same else np.where(labels != labels[idx1])[0]
        idx2 = np.random.choice(candidates)
        a.append(images[idx1]); b.append(images[idx2]); targets.append(int(same))
    return np.array(a), np.array(b), np.array(targets)

a, b, y = make_pairs(x_train, y_train)
inp1, inp2 = layers.Input((28,28)), layers.Input((28,28))
base = models.Sequential([layers.Flatten(), layers.Dense(64, activation="relu"),
                          layers.Dense(32, activation="relu")])
e1, e2 = base(inp1), base(inp2)
distance = layers.Lambda(lambda z: tf.sqrt(tf.reduce_sum(tf.square(z[0]-z[1]), axis=1, keepdims=True)+1e-7))([e1,e2])
out = layers.Dense(1, activation="sigmoid")(layers.Concatenate()([e1,e2, distance]))
model = models.Model([inp1, inp2], out)
model.compile(optimizer="adam", loss="binary_crossentropy", metrics=["accuracy"])
model.fit([a,b], y, epochs=3, batch_size=64, verbose=1)
print("Pair prediction (similarity score):", model.predict([a[:2], b[:2]], verbose=0).ravel())

# Demo only: generate simple 16x16 signature-like patterns
np.random.seed(2)
X = np.random.rand(300,16,16).astype("float32")
y = np.repeat(np.arange(10), 30)
for i in range(len(X)):
    X[i, 5:7, :] += y[i] / 20
X = np.clip(X, 0, 1)
a, b, same = [], [], []
for i in range(500):
    i1 = np.random.randint(len(X)); positive = i % 2 == 0
    choices = np.where(y == y[i1])[0] if positive else np.where(y != y[i1])[0]
    i2 = np.random.choice(choices)
    a.append(X[i1]); b.append(X[i2]); same.append(int(positive))

dist = layers.Lambda(lambda z: tf.reduce_mean(tf.abs(z[0]-z[1]), axis=1, keepdims=True))([e1,e2])
out = layers.Dense(1,activation="sigmoid")(layers.Concatenate()([e1,e2,dist]))
model = models.Model([i1,i2],out)
model.compile(optimizer="adam",loss="binary_crossentropy",metrics=["accuracy"])

import tensorflow as tf
from tensorflow.keras import layers, models
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

X,y = load_iris(return_X_y=True)

    for rate in [0.001, 0.01]:
        model = models.Sequential([layers.Input((4,)),layers.Dense(16,activation="relu"),
                                   layers.Dense(3,activation="softmax")])

                      loss="sparse_categorical_crossentropy",metrics=["accuracy"])
        model.fit(X_train,y_train,epochs=20,batch_size=batch,verbose=0)

              "accuracy=",round(model.evaluate(X_test,y_test,verbose=0)[1],3))
